\section*{Problem 1}

\subsection*{Problem Statement}
Let $a$ and $b$ be integers, Prove that if $2 + a$ and $35 - b$ are divisible by 11, then $a + b$ is divisible by 11.

\subsection*{Solution}

\begin{proof}
	Since $2 + a$ and $35 - b$ are divisible by 11, it can be represented with the following equations:

	\begin{align*}
		2 + a = 11 \cdot p, \qquad p \in \mathbb{Z} \\
		35 - b = 11 \cdot q, \qquad q \in \mathbb{Z}
	\end{align*}

	where $p$ and $q$ are integers.

	Each equations can be rearranged as follows.

	\begin{align*}
		a = 11 \cdot p - 2 \\
		b = 35 - 11 \cdot q
	\end{align*}

	The sum $a + b$ is then as follows.

	\begin{align*}
		a + b & = 11 p - 2 + 35 - 11q           &                              \\
		      & = 11 \cdot (p - q) + 33         &                              \\
		      & = 11 \cdot (p - q) + 11 \cdot 3 &                              \\
		      & = 11 \cdot (p - q + 3)          &                              \\
		      & = 11 \cdot k                    & k = p - q + 3 \in \mathbb{Z}
	\end{align*}

	$k$ is guaranteed to be an integer since it is a sum of integers.
	Since $a + b$ is a product of 11 and some integer $k$, $a + b$ is divisible by 11.

\end{proof}

\subsection*{Problem Statement}

Use Euclidean algorithm to find:
\begin{enumerate}[label=(\alph*)]
	\item (1003, 456)
	\item (5858, 1436)
\end{enumerate}

\subsection*{Solution}

\subsubsection*{(a)}

\begin{align*}
	(1003, 456) & = (456, 91) & 1003 & = 456 \times 2 + 91 \\
	            & = (91, 1)   & 456  & = 91 \times 5 + 1   \\
	            & = (1, 0)    & 91   & = 1 \times 91 + 0   \\
	            & = 1                                      \\
\end{align*}

Answer: $(1003, 456) = 1$

\subsubsection*{(b)}

\begin{align*}
	(5858, 1436) & = (1436, 114) & 5858 & = 1436 \times 4 + 114 \\
	             & = (114, 68)   & 1436 & = 114 \times 12 + 68  \\
	             & = (68, 46)    & 114  & = 64 \times 1 + 46    \\

	             & = (2, 0)      & 22   & = 2 \times 11 + 0     \\
	             & = 2
\end{align*}

Answer: $(5858, 1436) = 2$

\section*{Problem 4}

\subsection*{Problem Statement}

Show that for any positive integer $n$, the fraction $\frac{12 n + 1}{30n + 2}$ cannot be reduced.

\subsection*{Solution}

\begin{proof}
	For a fraction to be non-reducible, the denominator and the numerator must be coprime. In other words, the GCD of the denominator and the numerator must be 1.

	The problem may be reformulated as follows: show that for any positive integer $n$, $(12n + 1), 30n + 2$ is 1. Note that since $n$ is a positive integer, $12n + 1$ is always less than $30n + 2$.

	Applying the Euclidean algorithm results in the following:

	\begin{align*}
		(30n + 2, 12n + 1) & = (12n + 1, 6n) \\
		                   & = (6n, 1)       \\
		                   & = (1, 0)        \\
		                   & = 1
	\end{align*}

	From the above result, it is shown that the GCD of $30n + 2$ and $12n + 1$ is 1, meaning that $30n + 2$ and $12n + 1$ are coprime. Therefore, the fraction $\frac{12n + 1}{30n + 2}$ cannot be reduced for any positive integer $n$

\end{proof}

\section*{Problem 5}

\subsection*{Problem Statement}

Find $(2^{100} - 1, 2^{120} - 1)$.

\subsection*{Solution}

Let $q \in \mathbb{Z}^+$ be the GCD of $2^{100} - 1$ and $2^{120} - 1$. This can be represented as the following.

\begin{align*}
	(2^{100} - 1) \bmod q & \equiv 0 \bmod q \\
	(2^{120} - 1) \bmod q & \equiv 0 \bmod q
\end{align*}

By modulus arithmetic, the relations can be rearranged as following.

\begin{align*}
	2^{100} \bmod q \equiv 1 \bmod q \\
	2^{120} \bmod q \equiv 1 \bmod q
\end{align*}

Then, the two relations can be combined.

\begin{align*}
	2^{100} \bmod q \equiv 2^{120} \bmod q \\
\end{align*}

Divide both sides by $2^{100} \bmod q$ and rearrange.

\begin{align*}
	 & 2^{20} \bmod q \equiv 1 \bmod q       \\
	 & (2^{20} - 1) \bmod q \equiv 0 \bmod q
\end{align*}

$q$ must divide $2^{20} - 1$. The largest number that can achieve this is $2^{20} -1$. Therefore, the GCD of $(2^{100} - 1, 2^{120} - 1)$ is $2^{20} -1$.

\section*{Problem 6}

Textbook p16 (29)

\subsection*{Problem Statement}

Prove that a positive integer is divisible by 9 if and only if the sum of its digits is divisible by 9.

\subsection*{Solution}

First, we prove that if a positive integer is divisible by 9, its sum of digits is divisble by 9.

\begin{proof}
	Let $d_0, d_1, d_2, \ldots, d_{n-1}$ be the digits of a $n$ digit positive number $k$ where $k = d_0 + 10^1 \cdot d_1 + 10^2 \cdot d_2 + \ldots + 10^{n-1} \cdot d_{n-1}$. We would like to prove that if 9 divides $k$, 9 divides sum of $d_0, d_1, d_2, \ldots, d_{n-1}$.


	$k$ can be reformulated as follows:

	\begin{align*}
		k & = d_0 + 10^1 \cdot d_1 + 10^2 \cdot d_2 + \ldots + 10^{n-1} \cdot d_{n-1} \\
		  & = \sum_{i=0}^{n-1}{10^i \cdot d_i}                                        \\
		  & = \sum_{i=0}^{n-1}{d_i + (10^i - 1) \cdot d_i}                            \\
		  & = \sum_{i=0}^{n-1}{d_i} + \sum_{i=0}^{n-1}{(10^i - 1) \cdot d_i}          \\
	\end{align*}

	By factoring $10^i - 1$, $k$ can be further reformulated as follows:

	\begin{align*}
		k & = \sum_{i=0}^{n-1}{d_i} + \sum_{i=0}^{n-1}{(10^i - 1) \cdot d_i}                                          \\
		  & = \sum_{i=0}^{n-1}{d_i} + \sum_{i=0}^{n-1}{(10 - 1)(10^{i-1} + 10^{i-2} + \ldots + 10^{1} + 1) \cdot d_i} \\
		  & = \sum_{i=0}^{n-1}{d_i} + 9 \sum_{i=0}^{n-1}{(10^{i-1} + 10^{i-2} + \ldots + 10^{1} + 1) \cdot d_i}       \\
	\end{align*}

	Now $k$ is in the form $k = ab + r$, where $r = \sum_{i=0}^{n-1}{d_i}$, $a = 9$, and $b = \sum_{i=0}^{n-1}{(10^{i-1} + 10^{i-2} + \ldots + 10^{1} + 1) \cdot d_i}$. Since 9 divides $k$, 9 also must divide $r$, which is the sum of digits. Therefore, if a positive integer is divisible by 9, sum of its digits are also divisible by 9.
\end{proof}

Next, we prove that if the sum of a positive integer's digits is divisible by 9, the positive integer is also divisible by 9.

\begin{proof}
	From previous formulation of $k$, it is established that $k = ab + r$ where $a = 9$. Since $r$ is the sum of digits and can be represented in the form $r = 9p$ for some integer $p$, $k$ can be reformulated as follows:

	\begin{align*}
		k & = ab + r  \\
		  & = 9b + 9p \\
		  & = 9(b+p)
	\end{align*}

	This shows that $k$ is a multiple of 9, meaning that it is divisible by 9.
\end{proof}

\section*{Problem 7}

Textbook p16 (31.b)

\subsection*{Problem Statement}

If $t$ is an integer such that $a_i \mid t$ for $i = 1, 2, ... , k$, prove that $[a_1, a_2, \ldots, a_k] \mid t$.

\subsection*{Solution}

\begin{proof}
	Let $m = [a_1, a_2, \ldots, a_k]$. By division algorithm, $t = mq + r$. For $t$ to be divisible by $m$, $r$ must be 0.

	For every $i$, let $t = a_i u_i$ and $m = a_i v_i$ since $t$ and $m$ are divisible by $a_i$. By rearranging the equation found by division algorithm, $r = t - mq$. Substituting $t$ and $m$ to this equation results in the following: $r = a_i(u_i - v_i q)$. Therefore, $r$ is divisible by $a_i$ for all $i$
\end{proof}

\section*{Problem 8}

Textbook p23 (21)

\subsection*{Problem Statement}

If $c^2 = ab$ and $(a, b) = 1$, where $a, b, c$ are positive integers, prove that $a$ and $b$ are perfect squares.

\subsection*{Solution}

\begin{proof}
	Assume contradiction: if $c^2 = ab$ and $(a, b) = 1$, then $a$ and $b$ are not perfect squares.

	From the second equation, it can be deduced that $a$ and $b$ are coprime, meaning that $a$ does not equal to $b$.

	Square rooting the first equation yields the following equation: $c = \sqrt{ab} = \sqrt{a} \sqrt{b}$. Since $a$ does not equal to $b$, $a$ and $b$ multiplied cannot result in a perfect square. However, since $a$ and $b$ are positive integers, square root of them is irrational. This is a contradiction, as $c$ must also be a positive integer.

	Therefore, by contradiction, if $c^2 = ab$ and $(a, b) = 1$, $a$ and $b$ are perfect squares.
\end{proof}

\section*{Problem 9}

Textbook p24 (25)

\subsection*{Problem Statement}

Let $p$ be prime and $1 \leq k < p$. Prove that $p$ divides the binomial coefficient $\binom{p}{k}$.

\subsection*{Solution}

\begin{proof}
	Recall definition of binomial coefficient.

	$$
		\binom{p}{k} = \frac{p!}{k!(p - k)!}
	$$

	Rearrange the equation.

	$$
		p! = \binom{p}{k} k! (p - k)!
	$$

	The left-hand side is divisible by $p$. By Euclid's Lemma, one of two terms $\binom{p}{k}$ and $k! (p - k)!$ must be divisible by $p$. However, the second term is not divisible by $p$, since $k < p$ and $p - k < p$. Therefore, $\binom{p}{k}$ must be the term divisible by $p$.
\end{proof}
